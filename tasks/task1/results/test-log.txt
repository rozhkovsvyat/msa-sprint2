$ grep -n -e 'finalPrice = basePrice - discount' -e 'setDiscountPercent(discount)' src/main/java/com/hotelio/monolith/service/BookingService.java

58:        booking.setDiscountPercent(discount);

$ grep -n 'isVip ? 80.0 : 100.0' src/main/java/com/hotelio/monolith/service/BookingService.java
94:            log.debug("User {} has status '{}', base price is {}", userId, status, isVip ? 80.0 : 100.0);
